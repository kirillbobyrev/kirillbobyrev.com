{{- /* Figure shortcode. When `src` resolves to a bundled page resource it is processed (see responsive-img.html); otherwise the raw `src` (external or /static path) is used as-is. API: src, alt, caption, width, height, wide. `wide="true"` breaks the figure out of the prose measure (see .figure-wide in main.css) for images/diagrams that benefit from more horizontal room. */ -}} {{- $src := .Get "src" | default "" -}} {{- $alt := .Get "alt" | default (.Get "caption") -}} {{- $res := "" -}} {{- if and (ne $src "") (not (urls.Parse $src).IsAbs) (not (strings.HasPrefix $src "/")) }} {{- $res = .Page.Resources.Get $src -}} {{- end -}} {{- $class := "figure-block" -}} {{- if eq (.Get "wide") "true" -}} {{- $class = printf "%s figure-wide" $class -}} {{- end -}}
