{{- /* A responsive, lazily loaded from an image Resource. Params (dict): image (resource), alt (string). A WebP srcset capped at the original width, with intrinsic width/height against layout shift. The image shows at its natural width (blocks.css), so `sizes` is that width, or 92vw where the screen is narrower; the larger steps serve high-density screens. */ -}} {{- $res := .image -}} {{- $maxw := $res.Width -}} {{- $cap := cond (gt $maxw 2560) 2560 $maxw -}} {{- $widths := slice -}} {{- range (slice 480 768 1024 1440 2048) -}} {{- if lt . $maxw }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- $widths = $widths | append $cap | uniq | sort -}} {{- $srcset := slice -}} {{- range $widths -}} {{- $i := $res.Resize (printf "%dx webp q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $i.RelPermalink .) -}} {{- end -}} {{- $fallback := $res.Resize (printf "%dx webp q82" $cap) -}}
